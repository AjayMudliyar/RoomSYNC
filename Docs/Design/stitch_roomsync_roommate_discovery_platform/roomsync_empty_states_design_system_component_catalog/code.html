  primaryAction: { label: string; onClick: () => void; icon?: string };
  secondaryAction?: { label: string; onClick: () => void; icon?: string };

export const EmptyStateContainer: React.FC<EmptyStateProps> = ({ ...props }) => {
  return (
    <View style={[styles.canvas, { minHeight: 420 }]} accessibilityRole="alert">

  );
};
